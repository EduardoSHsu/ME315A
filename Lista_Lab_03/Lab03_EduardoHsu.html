getStats <- function(input, pos) {
  input %>% 
    filter(!str_starts(DESTINATION_AIRPORT, "1")) %>%
    drop_na() %>%
    group_by(DESTINATION_AIRPORT) %>%
    summarise(
    voos_total = n(),
    voos_atraso = sum(ARRIVAL_DELAY))
}
